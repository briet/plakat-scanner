Optional: Volle Offline-PWA (manifest.json + sw.js)
      
Dafür im gleichen Ordner wie die HTML-Datei zwei Dateien anlegen:

      
manifest.json:

      
{
  "name": "Plakat-Scanner",
  "short_name": "Plakat",
  "start_url": "./",
  "display": "standalone",
  "background_color": "#f1f5f9",
  "theme_color": "#0f172a",
  "icons": [{ "src": "icon.png", "sizes": "512x512", "type": "image/png" }]
}

      
sw.js:

      
self.addEventListener('install', e => self.skipWaiting());
self.addEventListener('fetch', e => {
  e.respondWith(caches.open('v1').then(async c => {
    const hit = await c.match(e.request);
    if (hit) return hit;
    const res = await fetch(e.request);
    if (res.ok && e.request.url.startsWith(self.location.origin)) c.put(e.request, res.clone());
    return res;
  }));
});

      
Und im <head> ergänzen:

      
<link rel="manifest" href="manifest.json">
<script>
if ('serviceWorker' in navigator && location.protocol === 'https:')
  navigator.serviceWorker.register('sw.js');
</script>
